        self.left = left
        self.right = right

        self.hash = self._compute_hash()
        self.lock = threading.Lock() # Conceptual substitute for atomic CAS propagation

    def _compute_hash(self):
        hasher = hashlib.sha256()
        if self.left: hasher.update(self.left.hash.encode())

        if self.data: hasher.update(self.data.encode())

            self.hash = self._compute_hash()
