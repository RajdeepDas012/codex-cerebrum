### Code Example
*Demonstrates a lock-free Fenwick Tree update utilizing atomic compare-and-swap operations on bitwise index masks.*

```cpp
#include <atomic>
#include <vector>

class ConcurrentFenwickTree {
    std::vector<std::atomic<long>> tree;
    size_t n;
public:
    ConcurrentFenwickTree(size_t size) : n(size), tree(size + 1) {
        for(auto& val : tree) val.store(0, std::memory_order_relaxed);
    }
    void add(size_t idx, long delta) {
        for (++idx; idx <= n; idx += idx & -idx) {
            long current = tree[idx].load(std::memory_order_relaxed);
            while (!tree[idx].compare_exchange_weak(current, current + delta,
                    std::memory_order_release, std::memory_order_relaxed)) {}
        }
    }
};
```

```cpp
}
```
