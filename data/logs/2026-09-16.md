```cpp
        size_t index = hash % num_registers;
        uint8_t leading_zeros = __builtin_clzll(hash | num_registers) + 1;
        
        uint8_t current = registers[index].load(std::memory_order_relaxed);
        while (leading_zeros > current && 
               !registers[index].compare_exchange_weak(current, leading_zeros, 
                   std::memory_order_release, std::memory_order_relaxed)) {
            // CAS loop repeats if current value changed
        }
    }
};
```

- ✅ Choose a power-of-two register count to allow fast bitwise masking instead of slow modulo operations during hashing index lookups.
