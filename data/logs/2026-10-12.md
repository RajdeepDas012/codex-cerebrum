bool refitNodeAtomic(BVHNode* node, const BBox& newBox, HazardPointerList& hpList, std::atomic<bool>& lockFlag) {
    HazardPointer hp = hpList.acquire(node);
    while (true) {
        BBox oldBox = node->box.load(std::memory_order_relaxed);
        BBox merged = oldBox.merge(newBox);
        if (node->box.compare_exchange_weak(oldBox, merged, std::memory_order_release, std::memory_order_relaxed)) {

            return true;
        }
    }
}
