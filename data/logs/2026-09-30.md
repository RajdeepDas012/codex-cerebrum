*Demonstrates a thread-safe SPSC ring buffer utilizing atomic operations and acquire-release memory ordering.*

```cpp
template <typename T, size_t Capacity>
class SPSCRingBuffer {
    struct Node {
        T data;
    };
    Node buffer[Capacity + 1];
    alignas(64) std::atomic<size_t> head{0};
    alignas(64) std::atomic<size_t> tail{0};
public:
    bool push(const T& item) {
        size_t current_head = head.load(std::memory_order_relaxed);
        size_t current_tail = tail.load(std::memory_order_acquire);
        if ((current_head + 1) % (Capacity + 1) == current_tail) return false;
        buffer[current_head].data = item;
        head.store((current_head + 1) % (Capacity + 1), std::memory_order_release);
        return true;
    }
    bool pop(T& item) {
        size_t current_tail = tail.load(std::memory_order_relaxed);
        size_t current_head = head.load(std::memory_order_acquire);
        if (current_tail == current_head) return false;
        item = buffer[current_tail].data;
        tail.store((current_tail + 1) % (Capacity + 1), std::memory_order_release);
        return true;
    }
};
```
