### Code Example
*Demonstrates thread-safe atomic state transitions and EMA failure rate updates for a circuit breaker.*

```python

import threading
from enum import Enum

class State(Enum):
    CLOSED = 1
    OPEN = 2
    HALF_OPEN = 3

class LockFreeCircuitBreaker:
    def __init__(self, failure_threshold=0.5, alpha=0.2, recovery_time=5.0):
        self.state = State.CLOSED
        self.failure_rate = 0.0
        self.alpha = alpha
        self.failure_threshold = failure_threshold
        self.recovery_time = recovery_time
        self.last_state_change = time.time()

        sample = 0.0 if success else 1.0

                self.state = State.OPEN
                self.last_state_change = time.time()
```
